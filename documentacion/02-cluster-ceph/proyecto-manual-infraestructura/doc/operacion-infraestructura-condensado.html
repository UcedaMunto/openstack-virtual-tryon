Script stop-all — detener toda la infraestructura en orden seguro

#!/bin/bash
# stop-all-services.sh  -  ejecutar desde host KVM
# Detiene servicios en orden inverso al arranque para evitar split-brain en Galera.

SSH_BASE="ssh -i ssh-keys/id_rsa -o StrictHostKeyChecking=no -o ConnectTimeout=8 userinfrakv"

run() { local ip=$1; shift; echo "[STOP] $ip: $*"; $SSH_BASE@$ip "sudo systemctl stop $*" 2>/dev/null && echo "  OK" || echo "  WARN (puede no estar corriendo)"; }

echo "=== 1. LBs (keepalived + nginx) ==="
run 192.168.10.20 keepalived nginx
run 192.168.10.21 keepalived nginx

echo "=== 2. Backend WinterCMS ==="
run 192.168.20.10 nginx php8.1-fpm
run 192.168.20.11 nginx php8.1-fpm
run 192.168.20.12 nginx php8.1-fpm

echo "=== 3. MaxScale ==="
run 192.168.30.20 maxscale

echo "=== 4. Redis (todos) ==="
for ip in 192.168.30.16 192.168.30.15 192.168.30.14 192.168.30.13 192.168.30.12 192.168.30.11 192.168.30.10; do
  run $ip redis-server
done

echo "=== 5. MariaDB Galera (orden: db3 -> db2 -> db1) ==="
run 192.168.30.23 mariadb
sleep 2
run 192.168.30.22 mariadb
sleep 2
run 192.168.30.21 mariadb

echo "=== 6. DNS ==="
run 192.168.10.11 bind9
run 192.168.10.10 bind9

echo "=== DONE - todos los servicios detenidos ==="

Como levantar un nuevo dominio en dns-delegado (192.168.10.11)

Dos casos: A) Subdominio de un dominio ya existente en dns-principal (ej. nuevo.mimas.net) — requiere delegar desde dns-principal.  |  B) Dominio completamente nuevo (ej. midominio.com) — solo se declara en el delegado, no necesita cambios en dns-principal salvo que quieras forward.

Paso 1 — Crear el archivo de zona en dns-delegado (192.168.10.11)

ssh -i ssh-keys/id_rsa userinfrakv@192.168.10.11 "sudo tee /etc/bind/db.midominio.com > /dev/null" <<'EOF'
; /etc/bind/db.midominio.com
; Reemplazar MIDOMINIO.COM y las IPs por los valores reales.
; Serial: formato AAAAMMDDNN  (ej. 2026051401 = 14 mayo 2026, revision 01)

\$TTL 300

@   IN  SOA  ns1.ti.mimas.net. admin.midominio.com. (
                2026051401   ; serial  <-- incrementar en cada cambio
                3600         ; refresh
                900          ; retry
                604800       ; expire
                300 )        ; negative TTL

; Servidor de nombres autoridad
@   IN  NS   ns1.ti.mimas.net.

; A record del NS (glue record)
ns1.ti.mimas.net. IN A 192.168.10.11

; ---- Registros del dominio ----
@       IN  A    192.168.10.20    ; midominio.com -> LB1
www     IN  A    192.168.10.20    ; www.midominio.com -> LB1
app1    IN  A    192.168.10.20    ; app1.midominio.com
; mail  IN  A    X.X.X.X         ; agregar si hay servidor de correo
EOF

Paso 2 — Declarar la zona en named.conf.local del delegado

ssh -i ssh-keys/id_rsa userinfrakv@192.168.10.11 "sudo tee -a /etc/bind/named.conf.local" <<'EOF'

// Nueva zona: midominio.com
zone "midominio.com" {
    type master;
    file "/etc/bind/db.midominio.com";
    allow-query { any; };
    // allow-transfer { none; };  // descomentar para bloquear AXFR
};
EOF

Paso 3 — Verificar sintaxis y recargar BIND en el delegado

# Validar el archivo de zona
ssh -i ssh-keys/id_rsa userinfrakv@192.168.10.11 \
  "sudo named-checkzone midominio.com /etc/bind/db.midominio.com"

# Validar la configuracion global
ssh -i ssh-keys/id_rsa userinfrakv@192.168.10.11 \
  "sudo named-checkconf && echo OK"

# Recargar BIND sin reiniciar (carga la nueva zona)
ssh -i ssh-keys/id_rsa userinfrakv@192.168.10.11 \
  "sudo rndc reconfig && sudo rndc reload midominio.com"

# Verificar que la zona esta activa
ssh -i ssh-keys/id_rsa userinfrakv@192.168.10.11 \
  "sudo rndc zonestatus midominio.com"

Paso 4 — (Solo caso A) Delegar desde dns-principal si es subdominio de mimas.net o gepardo.com

# Si el nuevo dominio es un subdominio de uno ya gestionado en dns-principal (10.10),
# hay que agregar registros NS + glue en el archivo de zona del padre.
# Ejemplo: nuevo.mimas.net delegado a 192.168.10.11

ssh -i ssh-keys/id_rsa userinfrakv@192.168.10.10 "sudo nano /etc/bind/db.mimas.net"
# Agregar al final del archivo:
#   nuevo.mimas.net.   IN  NS   ns1.ti.mimas.net.
#   ns1.ti.mimas.net.  IN  A    192.168.10.11      ; glue record (si no existe ya)

# Incrementar el serial del SOA en db.mimas.net, luego:
ssh -i ssh-keys/id_rsa userinfrakv@192.168.10.10 \
